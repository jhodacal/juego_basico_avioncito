public void PintarTexturaAvion(Graphics g)
{
    foreach (var capa in capasColores)
    {
        using (SolidBrush brush = new SolidBrush(capa.color))
        {
            foreach (var p in capa.puntos)
            {
                g.FillRectangle(brush, p.X, p.Y, 1, 1);
            }
        }
    }
}
